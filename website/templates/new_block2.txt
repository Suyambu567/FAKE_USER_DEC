        const pieCtx = document.getElementById('predictionDistributionChart').getContext('2d');
        const pieChart = new Chart(pieCtx, {
            type: 'pie',
            data: {
                labels: ['Fake Accounts', 'Real Accounts'],
                datasets: [{
                    data: [{{ fake_count }}, {{ real_count }}],
                    backgroundColor: [
                        'rgba(231, 76, 60, 0.8)',
                        'rgba(39, 174, 96, 0.8)'
                    ],
                    borderColor: [
                        'rgba(231, 76, 60, 1)',
                        'rgba(39, 174, 96, 1)'
                    ],
                    borderWidth: 2
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                plugins: {
                    legend: {
                        position: 'bottom',
                    },
                    tooltip: {
                        callbacks: {
                            label: function(context) {
                                const label = context.label || '';
                                const value = context.parsed || 0;
                                const sum = ctx.chart.data.datasets[0].data.reduce((a, b) => a + b, 0);
                                const percentage = ((value / sum) * 100).toFixed(1) + '%';
                                return label + ': ' + value + ' (' + percentage + ')';
                            }
                        }
                    }
                }
            }
        });

        const healthCtx = document.getElementById('systemHealthChart').getContext('2d');

                    legend: {

                    },
                    tooltip: {
                        callbacks: {
                            label: function(context) {
                                let value = context.parsed.x;
                                if (context.datasetIndex === 0) { // Accuracy
                                    return value + '%';
                                } else if (context.datasetIndex === 1) { // Response Time
                                    return value + 's';
                                } else if (context.datasetIndex === 2) { // Uptime
                                    return value + '%';
                                } else { // Memory Usage
                                    return value + '%';
                                }
                            }
                        }
                    }
                },
                scales: {
                    x: {
